\section*{7. Práva a povinnosti účastníků smlouvy}

\textbf{Pronajímatel je povinen a zavazuje se:}
\begin{itemize}
    \item předat nájemci byt ve stavu způsobilém k řádnému užívání, v tomto stavu ho svým nákladem udržovat, a zajistit nájemci plný a nerušený výkon práv spojených s užíváním bytu,
    \item bez zbytečného odkladu provést opravy, které nejsou opravami drobnými.
\end{itemize}

\textbf{Pronajímatel má právo:}
\begin{itemize}
    \item provádět stavební úpravy bytu a jiné podstatné změny v bytě pouze se souhlasem nájemce,
    \item provést na svůj náklad drobné opravy a běžnou údržbu bytu, nepostará-li se o jejich včasné provedení nájemce, a požadovat pak náhradu,
    \item odstranit na svůj náklad poškození, která na bytu způsobil nájemce, a požadovat náhradu.
\end{itemize}

\textbf{Nájemce je povinen a zavazuje se:}
\begin{itemize}
    \item užívat byt výhradně k bydlení,
    \item pečovat o to, aby na bytu nevznikla škoda,
    \item udržovat na svůj náklad byt v čistém a uživatelném stavu, provádět drobné opravy (do částky 1000,- Kč),
    \item udržovat el. spotřebiče, které nejsou součástí bytu, v provozuschopném stavu,
    \item umožnit pronajímateli přístup do bytu za účelem kontroly,
    \item že nedá byt do podnájmu třetí osobě,
    \item že nebude v bytě kouřit.
\end{itemize}
